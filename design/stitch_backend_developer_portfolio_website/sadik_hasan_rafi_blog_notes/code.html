  if (!acquired) {
    return { status: 200, detail: 'deduped_flight' };
  }

    // 2. Guard against replay via atomic record
    const exists = await tx.ledger.findUnique({
      where: { eventId: evt.id }
    });
    if (exists) return exists;

    return await tx.ledger.create({ ... });
  });
}
